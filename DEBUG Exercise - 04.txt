// This pseudocode segment is intended to compute and display
// the cost of home ownership for any number of users
// The program ends when a user enters 0 for the mortgage payment

//changed "input mtgPayment" to "input mortgagePayment"
//changed the startup() function call to startUp()
//changed the "MainLoop()" condition to mainLoop()
//added startUp() to prevent an infinite loop using only mainLoop()

// so after Declarations, startUp() will run once and read the first
// mortage payment.
// while the mortagePayment is not equal to 0, mainLoop() will
// collect utilities, taxes, and upkeep, Compute the totals and display it,
// then startUp() runs again to the next mortage payment
// when a user enters 0, the loop ends and finishUp() prints the end message.

start
   Declarations
      num mortgagePayment
      num utilities
      num taxes
      num upkeep
      num total
   startUp()
   while mortgagePayment not equal to 0
      mainLoop()
      startUp()
   endwhile
   finishUp()

startUp()
   output "Enter your mortgage payment or 0 to quit"
   input mortgagePayment
return

mainLoop()
   output "Enter utilities"
   input utilities
   output "Enter taxes"
   input taxes
   output "Enter amount for upkeep"
   input upkeep
   total = mortgagePayment + utilities + taxes + upkeep
   output "Total is ", total
return

finishUp()
   output "End of program"
return
